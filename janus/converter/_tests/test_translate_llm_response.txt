            IYH_EDGEBUFFER = IGN_MASK['NROWS'] - int(EDGEBUFFER / IGN_MASK['CELLSIZE'])

                if float(IROW) * IGN_MASK['CELLSIZE'] < EDGEBUFFER:

            YSPAN = float(IGN_MASK['NROWS']) * IGN_MASK['CELLSIZE'] - 2.0 * EDGEBUFFER
            XLL = IGN_MASK['XLLCORNER'] + EDGEBUFFER
            YLL = IGN_MASK['YLLCORNER'] + EDGEBUFFER

            N = 0
            for IROW in range(IYL_EDGEBUFFER, IYH_EDGEBUFFER + 1):
                for ICOL in range(IXL_EDGEBUFFER, IXH_EDGEBUFFER + 1):
                    if IGN_MASK['R4'][ICOL - 1, IROW - 1, 0] < IGN_MASK_CRIT:
                        continue
                    if ISNONBURNABLE(ICOL, IROW) and not ALLOW_NONBURNABLE_PIXEL_IGNITION:
                        continue
                    N += 1

            ICOL_ARR = np.zeros(NUM_IGNITABLE_PIXELS, dtype=int)
            IROW_ARR = np.zeros(NUM_IGNITABLE_PIXELS, dtype=int)
            IGN_MASK_ARR = np.zeros(NUM_IGNITABLE_PIXELS)
            PROB = np.zeros(NUM_IGNITABLE_PIXELS + 1)

            for IROW in range(IYL_EDGEBUFFER, IYH_EDGEBUFFER + 1):
                for ICOL in range(IXL_EDGEBUFFER, IXH_EDGEBUFFER + 1):

                    N_ARR[N - 1] = N
                    ICOL_ARR[N - 1] = ICOL
                    IROW_ARR[N - 1] = IROW

        IWX_SERIAL_BAND_TEMP = np.zeros(NUM_CASES_TOTAL_MAX, dtype=int)

                IX_IGNFAC[ICOL - 1] = min(max(ic_from_x(X, IGNFAC['XLLCORNER'], IGNFAC['CELLSIZE']), 1), IGNFAC['NCOLS'])

            IY_IGNFAC = np.zeros(IGN_MASK['NROWS'], dtype=int)

                Y = y_from_irow(IROW, IGN_MASK['YLLCORNER'], IGN_MASK['CELLSIZE'])
                IY_IGNFAC[IROW - 1] = min(max(ir_from_y(Y, IGNFAC['YLLCORNER'], IGNFAC['CELLSIZE']), 1), IGNFAC['NROWS'])

        for IWX_BAND in range(IWX_BAND_START, IWX_BAND_STOP + 1, IWX_BAND_SKIP):
            NUM_STARTING_WX_BANDS += 1

            if RANDOM_IGNITIONS_TYPE == 1:
                RCOUNT = float(NUM_IGNITABLE_PIXELS)
            elif RANDOM_IGNITIONS_TYPE == 2:
                RCOUNT = 0.0

                                RCOUNT_NOERC += IGNITION_MASK_SCALE_FACTOR * IGNITION_DENSITY_ADJ_PYROME(IPYROME) * IGN_MASK['R4'][ICOL - 1, IROW - 1, 0]
                            else:
                                RCOUNT_NOERC += IGNITION_MASK_SCALE_FACTOR * IGN_MASK['R4'][ICOL - 1, IROW - 1, 0]
                        RCOUNT_NOERC_SUM = RCOUNT_NOERC
                    else:
                        RCOUNT_NOERC_SUM += RCOUNT_NOERC

                        IROW = IROW_ARR[I - 1]
                        ICOL = ICOL_ARR[I - 1]
                        if USE_PYROMES and CALIBRATION_CONSTANTS_BY_PYROME:
                            IPYROME = min(max(PYROMES['I2'][ICOL - 1, IROW - 1, 0], 1), 128)
                            RCOUNT += IGNITION_MASK_SCALE_FACTOR * IGNITION_DENSITY_ADJ_PYROME(IPYROME) * IGN_MASK['R4'][ICOL - 1, IROW - 1, 0] * IGNFAC['R4'][IX_IGNFAC[ICOL - 1], IY_IGNFAC[IROW - 1], IWX_BAND]
                        else:

                    RCOUNT_SUM += RCOUNT
                else:
                    for I in range(1, NUM_IGNITABLE_PIXELS + 1):
                        IROW = IROW_ARR[I - 1]
                        ICOL = ICOL_ARR[I - 1]
                        if IGN_MASK['R4'][ICOL - 1, IROW - 1, 0] > 0.0:

                                IPYROME = min(max(PYROMES['I2'][ICOL - 1, IROW - 1, 0], 1), 128)
                                RCOUNT += IGNITION_MASK_SCALE_FACTOR * IGNITION_DENSITY_ADJ_PYROME(IPYROME) * IGN_MASK['R4'][ICOL - 1, IROW - 1, 0]
                            else:
                                RCOUNT += IGNITION_MASK_SCALE_FACTOR * IGN_MASK['R4'][ICOL - 1, IROW - 1, 0]

            else:
                print('Error: Set RANDOM_IGNITIONS_TYPE to 1 or 2')
                raise ValueError("Invalid RANDOM_IGNITIONS_TYPE")

            if NUM_ENSEMBLE_MEMBERS0 < 0:
                RNUM_ENSEMBLE_MEMBERS = 0.01 * PERCENT_OF_PIXELS_TO_IGNITE * RCOUNT
                FLOOR_NUM_ENSEMBLE_MEMBERS = np.floor(RNUM_ENSEMBLE_MEMBERS)
                FRAC = RNUM_ENSEMBLE_MEMBERS - float(FLOOR_NUM_ENSEMBLE_MEMBERS)
                R0 = np.random.rand()
                if R0 <= FRAC:
                    NUM_ENSEMBLE_MEMBERS = FLOOR_NUM_ENSEMBLE_MEMBERS + 1
                else:
                    NUM_ENSEMBLE_MEMBERS = FLOOR_NUM_ENSEMBLE_MEMBERS
                if NUM_ENSEMBLE_MEMBERS != NUM_ENSEMBLE_MEMBERS:
                    NUM_ENSEMBLE_MEMBERS = 0
            else:
                if NUM_ENSEMBLE_MEMBERS > round(RCOUNT) and not ALLOW_MULTIPLE_IGNITIONS_AT_A_PIXEL:
                    print('Error:  NUM_ENSEMBLE_MEMBERS is greater than the total number of ignitable pixels')
                    raise ValueError("Invalid NUM_ENSEMBLE_MEMBERS")

            if NUM_ENSEMBLE_MEMBERS > 0:
                NUM_CASES_PER_STARTING_WX_BAND[IWX_BAND - IWX_BAND_START] = NUM_ENSEMBLE_MEMBERS
                I1 = NUM_CASES_TOTAL + 1

def determine_ignition_locations():
    global STATS_IWX_BAND_START, STATS_IWX_SERIAL_BAND, STATS_X, STATS_Y, STATS_PROB
    STATS_IWX_BAND_START = IWX_BAND_START_TEMP[:NUM_CASES_TOTAL]
    del IWX_BAND_START_TEMP

    STATS_IWX_SERIAL_BAND = IWX_SERIAL_BAND_TEMP[:NUM_CASES_TOTAL]
    del IWX_SERIAL_BAND_TEMP

    ICASE = 0
    if USE_ERC:
        for IWX_BAND in range(IWX_BAND_START, IWX_BAND_STOP + 1, IWX_BAND_SKIP):
            if NUM_CASES_PER_STARTING_WX_BAND[IWX_BAND - IWX_BAND_START] < 1:
                continue

            IGN_MASK_SUM = 0.0
            for I in range(1, NUM_IGNITABLE_PIXELS + 1):
                IROW = IROW_ARR[I - 1]
                ICOL = ICOL_ARR[I - 1]
                IGN_MASK_ARR[I - 1] = IGN_MASK['R4'][ICOL - 1, IROW - 1, 0] * IGNFAC['R4'][IX_IGNFAC[ICOL - 1], IY_IGNFAC[IROW - 1], IWX_BAND]
                IGN_MASK_SUM += IGN_MASK_ARR[I - 1]

            IGN_MASK_ARR[:] /= IGN_MASK_SUM

            PROB = np.zeros(NUM_IGNITABLE_PIXELS + 1)
            for I in range(2, NUM_IGNITABLE_PIXELS + 1):
                PROB[I] = PROB[I - 1] + IGN_MASK_ARR[I - 1]
            PROB[NUM_IGNITABLE_PIXELS] = 1.0

            for ICASE_FOR_WX_BAND in range(1, NUM_CASES_PER_STARTING_WX_BAND[IWX_BAND - IWX_BAND_START] + 1):
                ICASE += 1

                R0 = np.random.rand()
                I = np.searchsorted(PROB, R0) - 1
                I = min(max(1, I), NUM_IGNITABLE_PIXELS)
                STATS_X[ICASE - 1] = x_from_icol(ICOL_ARR[I - 1], IGN_MASK['XLLCORNER'], IGN_MASK['CELLSIZE'])
                STATS_Y[ICASE - 1] = y_from_irow(IROW_ARR[I - 1], IGN_MASK['YLLCORNER'], IGN_MASK['CELLSIZE'])

                if RANDOM_IGNITIONS_TYPE == 1:
                    STATS_PROB[ICASE - 1] = IGN_MASK['R4'][ICOL_ARR[I - 1], IROW_ARR[I - 1], 0]
                else:
                    STATS_PROB[ICASE - 1] = 1.0
    else:
        IGN_MASK_SUM = 0.0
        for I in range(1, NUM_IGNITABLE_PIXELS + 1):
            IROW = IROW_ARR[I - 1]
            ICOL = ICOL_ARR[I - 1]
            IGN_MASK_ARR[I - 1] = IGN_MASK['R4'][ICOL - 1, IROW - 1, 0]
            IGN_MASK_SUM += IGN_MASK_ARR[I - 1]

        IGN_MASK_ARR[:] /= IGN_MASK_SUM

        PROB = np.zeros(NUM_IGNITABLE_PIXELS + 1)
        for I in range(2, NUM_IGNITABLE_PIXELS + 1):
            PROB[I] = PROB[I - 1] + IGN_MASK_ARR[I - 1]
        PROB[NUM_IGNITABLE_PIXELS] = 1.0

        for ICASE in range(1, NUM_CASES_TOTAL + 1):
            R0 = np.random.rand()
            I = np.searchsorted(PROB, R0) - 1
            I = min(max(1, I), NUM_IGNITABLE_PIXELS)
            STATS_X[ICASE - 1] = x_from_icol(ICOL_ARR[I - 1], IGN_MASK['XLLCORNER'], IGN_MASK['CELLSIZE'])
            STATS_Y[ICASE - 1] = y_from_irow(IROW_ARR[I - 1], IGN_MASK['YLLCORNER'], IGN_MASK['CELLSIZE'])

            if RANDOM_IGNITIONS_TYPE == 1:
                STATS_PROB[ICASE - 1] = IGN_MASK['R4'][ICOL_ARR[I - 1], IROW_ARR[I - 1], 0]
            else:
                STATS_PROB[ICASE - 1] = 1.0

    del N_ARR, ICOL_ARR, IROW_ARR, IGN_MASK_ARR, PROB
    if USE_ERC:
        del IX_IGNFAC, IY_IGNFAC

def write_cases_to_run():
